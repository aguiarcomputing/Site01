#!/bin/bash

# Script para verificar o status do firewall do Proxmox e listar regras ativas
# Execute como root

echo "=== Verificação do Firewall do Proxmox VE ==="
echo "Data: $(date)"
echo "Nó: $(hostname)"
echo ""

# 1. Verificar se o firewall está habilitado
echo "1. Status Geral do Firewall"
echo "---------------------------"
if systemctl is-active --quiet pve-firewall; then
    echo "Firewall do Proxmox: ATIVO"
else
    echo "Firewall do Proxmox: DESATIVADO"
    echo "Ação sugerida: Habilite via 'pve-firewall start' ou no GUI (Datacenter > Firewall > Options)"
fi
echo ""

# 2. Listar regras de firewall globais (Datacenter)
echo "2. Regras Globais (Datacenter)"
echo "---------------------------"
if [ -f /etc/pve/firewall/cluster.fw ]; then
    echo "Regras encontradas em /etc/pve/firewall/cluster.fw:"
    cat /etc/pve/firewall/cluster.fw | grep -v '^#' | grep -v '^$'
else
    echo "Nenhuma regra global configurada em /etc/pve/firewall/cluster.fw"
fi
echo ""

# 3. Listar regras de firewall do nó local
echo "3. Regras do Nó ($(hostname))"
echo "---------------------------"
NODE_FW="/etc/pve/nodes/$(hostname)/host.fw"
if [ -f "$NODE_FW" ]; then
    echo "Regras encontradas em $NODE_FW:"
    cat "$NODE_FW" | grep -v '^#' | grep -v '^$'
else
    echo "Nenhuma regra específica configurada para o nó em $NODE_FW"
fi
echo ""

# 4. Listar regras de firewall das VMs/CTs
echo "4. Regras de VMs e Containers"
echo "---------------------------"
for vm in /etc/pve/firewall/*.fw; do
    if [ -f "$vm" ]; then
        VMID=$(basename "$vm" .fw)
        echo "Regras para VM/CT ID $VMID:"
        cat "$vm" | grep -v '^#' | grep -v '^$'
        echo ""
    else
        echo "Nenhuma regra de firewall configurada para VMs/CTs"
    fi
done
echo ""

# 5. Listar regras ativas no iptables (nível do sistema)
echo "5. Regras Ativas no iptables"
echo "---------------------------"
echo "Regras INPUT:"
iptables -L INPUT -v -n --line-numbers
echo ""
echo "Regras OUTPUT:"
iptables -L OUTPUT -v -n --line-numbers
echo ""
echo "Regras FORWARD (para VMs/CTs):"
iptables -L FORWARD -v -n --line-numbers
echo ""

# 6. Verificar logs recentes do firewall
echo "6. Logs Recentes do Firewall (últimas 10 linhas)"
echo "---------------------------"
journalctl -u pve-firewall -n 10 --no-pager
echo ""

# 7. Testar compilação das regras
echo "7. Teste de Compilação das Regras"
echo "---------------------------"
pve-firewall compile
if [ $? -eq 0 ]; then
    echo "Compilação das regras: SUCESSO"
else
    echo "Erro na compilação das regras. Verifique configurações!"
fi
echo ""

echo "=== Fim do Relatório ==="
echo "Para mais detalhes, acesse o GUI (Datacenter > Firewall) ou verifique /etc/pve/firewall/"
