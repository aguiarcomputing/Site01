@ECHO OFF

echo Cria Sessão no Servidor Olld
net use \\xx.xx.xx.254 /USER:Administrador SENHA

echo Cria mapeamento
net use X: \\xx.xx.xx.254\c\Users /Y

echo Aguarda 2 minutos antes de iniciar as cópias
timeout /t 20

echo Acessando Mapemento
X:

echo copia dados perfil do usuário
robocopy X:\%UserName%\Desktop %HOMEPATH%\Desktop /MIR /W:0 /R:0 
robocopy X:\%UserName%\Documents %HOMEPATH%\Documents /MIR /W:0 /R:0 
robocopy X:\%UserName%\Downloads %HOMEPATH%\Downloads /MIR /W:0 /R:0 
robocopy X:\%UserName%\Favorites %HOMEPATH%\Favorites /MIR /W:0 /R:0 
robocopy X:\%UserName%\Links %HOMEPATH%\Links /MIR /W:0 /R:0 
robocopy X:\%UserName%\Pictures %HOMEPATH%\Pictures /MIR /W:0 /R:0 
robocopy X:\%UserName%\Videos %HOMEPATH%\Videos /MIR /W:0 /R:0

echo copia dados assinatura de email do usuário
robocopy X:\%UserName%\AppData\Roaming\Microsoft\Signatures %HOMEPATH%\AppData\Roaming\Microsoft\Signatures  /MIR /ZB /W:0 /R:0 

echo copia dados cache de contatos email enviados do usuário
robocopy X:\%UserName%\AppData\Local\Microsoft\Outlook\RoamCache %HOMEPATH%\AppData\Local\Microsoft\Outlook\RoamCache  /MIR /ZB /W:0 /R:0

echo copia atalhos da barra do menu iniciar do usuário
robocopy "X:\%UserName%\Appdata\Roaming\Microsoft\Internet Explorer\Quick Launch" "%HOMEPATH%\Appdata\Roaming\Microsoft\Internet Explorer\Quick Launch"  /MIR /ZB /W:0 /R:0 

echo remove mapeamento
net use X: /DELETE /y
